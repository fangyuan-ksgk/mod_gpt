# Undersized model experiments: does SoRL converge when baseline fails?
# Three architectures below the current 2L/3H/510d:
#   1L/3H/510d (3.91M transformer) — half the depth
#   1L/2H/256d (1.05M transformer) — shallow + narrow
#   2L/1H/128d (0.53M transformer) — deeper but very narrow

# ── 1L/3H/510d ──
python -m arithmetic.train --mode baseline --ops add_sub --dataset_size 500000 --num_epochs 10 --n_layer 1 --n_head 3 --n_embd 510 --push_to_hub --no_wandb --output_dir ckpt/sweep/as_baseline_500K_1L3H510d
python -m arithmetic.train --mode sorl --ops add_sub --dataset_size 500000 --abs_vocab 10 --K 1 --num_epochs 10 --n_layer 1 --n_head 3 --n_embd 510 --push_to_hub --no_wandb --output_dir ckpt/sweep/as_sorl_abs10_K1_500K_1L3H510d
